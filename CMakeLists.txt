    set(CMAKE_CONFIGURATION_TYPES ${BUILD_CONFIGS} CACHE STRING "" FORCE)

    if(NOT CMAKE_BUILD_TYPE)
        list(GET BUILD_CONFIGS 0 default_config)
        set(CMAKE_BUILD_TYPE ${default_config} CACHE STRING "One of ${BUILD_CONFIGS}" FORCE)
    endif()
    set_property(CACHE CMAKE_BUILD_TYPE PROPERTY STRINGS ${BUILD_CONFIGS})
    string(TOUPPER "${BUILD_CONFIGS}" configs_upper)

    if(NOT build_type_upper IN_LIST configs_upper)
        message(FATAL_ERROR "CMAKE_BUILD_TYPE '${CMAKE_BUILD_TYPE}' is none of ${BUILD_CONFIGS}.")
    endif()
endif()

function(study_source path)
    cmake_parse_arguments(PARSE_ARGV 1 ARG "OPTIMIZED" "DIALECT" "WHEN")
    list(JOIN ARGV " " entry)
    set(parsed 1)
    if(ARG_OPTIMIZED)
        math(EXPR parsed "${parsed} + 1")
    endif()
    if(DEFINED ARG_DIALECT)
        math(EXPR parsed "${parsed} + 2")
    endif()
    if(DEFINED ARG_WHEN)
        list(LENGTH ARG_WHEN conditions)
        math(EXPR parsed "${parsed} + 1 + ${conditions}")
    endif()
    if(NOT ARGC EQUAL parsed OR NOT path MATCHES "^[A-Za-z0-9_+-]+/[A-Za-z0-9_+-]+\\.c$"
       OR (DEFINED ARG_DIALECT AND NOT ARG_DIALECT IN_LIST KNOWN_DIALECTS))
        message(FATAL_ERROR "Malformed entry: study_source(${entry})")
    endif()
    foreach(condition IN LISTS ARG_WHEN)
        if(NOT condition MATCHES "^[A-Za-z_][A-Za-z0-9_]*$")
            message(FATAL_ERROR "Malformed entry: study_source(${entry})")
        endif()
        if(NOT DEFINED CACHE{${condition}})
            message(FATAL_ERROR "Unknown condition ${condition} in study_source(${entry})")
        endif()
    endforeach()

    if(DEFINED ARG_DIALECT AND NOT ARG_DIALECT IN_LIST C_DIALECTS)
        message(STATUS "      - skipped: ${path} (needs ${ARG_DIALECT} dialect)")
        return()
    endif()
    foreach(condition IN LISTS ARG_WHEN)
        set(met "$CACHE{${condition}}")
        if(NOT met)
            message(STATUS "      - skipped: ${path} (needs ${condition})")
            return()
        endif()
    endforeach()

    get_filename_component(dir "${path}" DIRECTORY)
    get_filename_component(name "${path}" NAME_WLE)
    add_executable(${dir}_${name} "${path}")
    set_target_properties(${dir}_${name} PROPERTIES
        OUTPUT_NAME ${name}
        FOLDER ${dir}
        RUNTIME_OUTPUT_DIRECTORY "${BIN_DIR}/${dir}"
        PDB_OUTPUT_DIRECTORY "${BIN_DIR}/${dir}"
    )
    set(note "")
    if(ARG_OPTIMIZED)
        target_compile_options(${dir}_${name} PRIVATE ${OPTIMIZED_FLAGS})
        list(JOIN OPTIMIZED_FLAGS " " note)
        set(note " (OPTIMIZED: ${note})")
    endif()
    message(STATUS "      + ${path}${note}")
endfunction()
